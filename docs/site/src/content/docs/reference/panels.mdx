import data from "../../../data/panel-reference.json";

Every panel that ships with the trading UI is listed here, each with a current screenshot pulled from a live render. Screenshots are auto-generated by the panel-walkthrough Playwright spec (`frontend/tests/panel-walkthrough.spec.ts`) on every CI run against `main` and committed back to `docs/panel-walkthrough/`. The four trader-only panels at the end have no screenshot yet. The walkthrough drives the admin role, which cannot open them. A separate trader walkthrough is planned (see follow-up).

For trading-style access (which roles can see which panels), see the [Trading styles](/veta-trading-platform/reference/trading-styles/) matrix.

<div class="panel-reference-grid">
  {data.panels.map((panel) => (
    <article class={`panel-card ${panel.rendered ? "is-rendered" : "is-trader-only"}`} id={`panel-${panel.panelId}`}>
      <header class="panel-card-header">
        <h2 class="panel-card-title">{panel.title}</h2>
        <code class="panel-card-id">{panel.panelId}</code>
        {!panel.rendered && (
          <span class="panel-card-badge">Trader-only</span>
        )}
      </header>
      {panel.description && (
        <p class="panel-card-description">{panel.description}</p>
      )}
      {panel.rendered ? (
        <a class="panel-card-figure" href={`/veta-trading-platform/${panel.screenshot}`}>
          <img
            src={`/veta-trading-platform/${panel.screenshot}`}
            alt={`${panel.title} screenshot`}
            loading="lazy"
            decoding="async"
          />
        </a>
      ) : (
        <div class="panel-card-placeholder">
          <p><strong>No screenshot.</strong> This panel is gated behind a trader-style permission and is not reachable from the admin walkthrough that produces these images.</p>
          <p>Notes from the walkthrough run: <em>{panel.notes ?? "skipped"}</em>.</p>
          <p class="panel-card-placeholder-followup">Follow-up: a trader-role walkthrough is planned to capture {data.skipped} additional panels.</p>
        </div>
      )}
    </article>
  ))}
</div>
